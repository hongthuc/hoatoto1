        let currentSlideIndex = 0;
        let slideInterval = null;

        function showSlide(index) {
            const slides = document.querySelectorAll('.hero-slide');
            const dots = document.querySelectorAll('.slide-dot');
            if (!slides.length) return;

            currentSlideIndex = (index + slides.length) % slides.length;
            slides.forEach((s, i) => {
                if (i === currentSlideIndex) {
                    s.classList.remove('hidden', 'opacity-0');
                    s.classList.add('opacity-100');
                } else {
                    s.classList.add('hidden', 'opacity-0');
                    s.classList.remove('opacity-100');
                }
            });

            dots.forEach((d, i) => {
                if (i === currentSlideIndex) {
                    d.className = "slide-dot w-6 h-2 rounded-full bg-toto-600 transition-all";
                } else {
                    d.className = "slide-dot w-2 h-2 rounded-full bg-white/80 hover:bg-white transition-all";
                }
            });
        }

        function changeSlide(step) {
            showSlide(currentSlideIndex + step);

        function goToSlide(index) {

        function resetSlideTimer() {
            if (slideInterval) clearInterval(slideInterval);
            slideInterval = setInterval(() => {
                showSlide(currentSlideIndex + 1);
            }, 4500);
        }
